async function initDashboard() { try { // 使用快取破壞參數 (Cache Buster)，強迫瀏覽器讀取最新的 data.json const cacheBuster = `?v=${Date.now()}_${Math.random().toString(36).substring(2, 7)}`; // 【已修正】補齊正確的語法結構與逗號 const response = await fetch('/-1/data.json' + cacheBuster, { headers: { 'Cache-Control': 'no-cache, no-store, must-revalidate' } }); if (!response.ok) throw new Error(`HTTP 錯誤! 狀態碼: ${response.status}`); const data = await response.json(); const setSafeText = (id, text) => { const el = document.getElementById(id); if (el) el.innerText = text; }; const setSafeHTML = (id, html) => { const el = document.getElementById(id); if (el) el.innerHTML = html; }; setSafeText('api-status', '✓ 數據即時更新成功'); const apiStatus = document.getElementById('api-status'); if (apiStatus) apiStatus.style.color = 'var(--green-down)'; // --- 1. 即時指標解析 --- const real = data.realtime || data.indicators || data || {}; const basis = Number(real.basis ?? real.price_basis ?? 0); const vol = Number(real.estimated_volume ?? real.volume ?? 0); const gap = Number(real.night_gap ?? real.gap ?? 0); const elec = Number(real.elec_share ?? real.elec_pct ?? 0); setSafeText('basis-val', `${basis >= 0 ? '+' : ''}${basis} 點`); setSafeText('basis-badge', basis >= 0 ? '正價差' : '逆價差'); const basisBadge = document.getElementById('basis-badge'); if (basisBadge) basisBadge.className = `badge ${basis >= 0 ? 'badge-green' : 'badge-red'}`; setSafeText('volume-val', vol ? `${vol} 億` : '--'); if (vol) setSafeText('volume-badge', vol > 3800 ? '爆量' : '縮量'); setSafeText('gap-val', `${gap >= 0 ? '+' : ''}${gap} 點`); setSafeText('gap-badge', gap >= 0 ? '高開' : '低開'); setSafeText('elec-pct', `${elec}%`); const elecBar = document.getElementById('elec-bar'); if (elecBar) elecBar.style.width = `${Math.min(elec, 100)}%`; // --- 2. 大盤與 Pivot 關卡 --- const t = data.taiex || data.index || data || {}; const price = Number(t.price ?? t.close ?? t.current_price ?? 0); const change = Number(t.change ?? 0); const changePct = Number(t.change_pct ?? t.change_percent ?? 0); setSafeText('data-date', t.date || new Date().toLocaleTimeString()); setSafeText('main-price', price ? price.toLocaleString() : '--'); const changeEl = document.getElementById('main-change'); if (changeEl) { const sign = change >= 0 ? '+' : ''; changeEl.innerText = `${sign}${change} (${sign}${changePct}%)`; changeEl.style.color = change >= 0 ? 'var(--red-up)' : 'var(--green-down)'; } setSafeText('val-high', t.high ? Number(t.high).toLocaleString() : '--'); setSafeText('val-low', t.low ? Number(t.low).toLocaleString() : '--'); setSafeText('val-prev', t.close_prev ? Number(t.close_prev).toLocaleString() : '--'); const p = t.pivot || data.pivot || {}; const s2 = Number(p.s2 ?? 0), s1 = Number(p.s1 ?? 0), pivotP = Number(p.p ?? 0), r1 = Number(p.r1 ?? 0), r2 = Number(p.r2 ?? 0); setSafeText('val-s2', s2 ? s2.toLocaleString() : '--'); setSafeText('val-s1', s1 ? s1.toLocaleString() : '--'); setSafeText('val-p', pivotP ? pivotP.toLocaleString() : '--'); setSafeText('val-r1', r1 ? r1.toLocaleString() : '--'); setSafeText('val-r2', r2 ? r2.toLocaleString() : '--'); if (pivotP > 0) { renderPivotChart({ s2, s1, p: pivotP, r1, r2 }, price); } // --- 3. AI 策略建議 --- const ai = data.ai_analysis || data.ai || {}; setSafeHTML('ai-alert-banner', `⚠️ AI 盤勢提醒： ${ai.alert_banner || ai.alert || '今日盤勢區間震盪，請注意風控'}`); setSafeHTML('ai-summary-text', ai.summary || ai.description || '目前數據分析完成。'); setSafeText('ai-scenario-a', ai.scenario_a || ai.plan_a || '突破站穩 P 點追多'); setSafeText('ai-scenario-b', ai.scenario_b || ai.plan_b || '跌破 S1 關卡停損'); // --- 4. 籌碼與 VPVR 圖表 --- if (data.chips && typeof data.chips === 'object') { renderChipsCharts(data.chips); } if (data.vpvr && typeof data.vpvr === 'object') { renderVPVRChart(data.vpvr); } } catch (err) { console.error("解析 JSON 時發生錯誤:", err); const apiStatus = document.getElementById('api-status'); if (apiStatus) { apiStatus.innerText = '❌ 讀取失敗，請確認 data.json 格式或 GitHub Pages 更新狀態'; apiStatus.style.color = 'var(--red-up)'; } } }
